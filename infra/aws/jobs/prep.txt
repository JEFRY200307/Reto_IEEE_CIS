# PREPARACIÓN ÚNICA (instancia CPU, con internet). ~45-60 min, < USD 0,50.
#   infra/aws/push.sh prep
#   infra/aws/launch.sh prep --template prep --keep-alive --max-hours 2
# Al terminar (infra/aws/pull.sh prep muestra SUMMARY.txt):
#   infra/aws/ami.sh <instance-id>
# bootstrap.sh ya creó /opt/dsqa-venv y descargó OLMo desde Hugging Face (no estaba en S3).

# 1. Modelos auxiliares de TODAS las estrategias (bge-base, bge-reranker, MiniLM para E0, Docling para E0) → S3
python offline/download_models.py --what embedder reranker minilm docling
aws s3 sync models/ "s3://$DSQA_BUCKET/models/" --only-show-errors

# 2. PDFs públicos del banco local (eval/bench/pdf_list.csv) → S3
python offline/fetch_public_pdfs.py --list eval/bench/pdf_list.csv
aws s3 sync data/public_pdfs/ "s3://$DSQA_BUCKET/corpus/" --only-show-errors

# 3. Glosario de DONES ampliado con esos PDFs (se guarda en la sesión: copiarlo a resources/ en la laptop)
mkdir -p "experiments/runs/$DSQA_SESSION" && cp resources/glossary_dones.json "experiments/runs/$DSQA_SESSION/glossary_dones.json" && python offline/build_glossary.py --pdfs data/public_pdfs/DONES --out "experiments/runs/$DSQA_SESSION/glossary_dones.json"

# 4. Caché de Hugging Face completa (OLMo + auxiliares) → S3: las sesiones GPU la descargan de ahí
aws s3 sync "$HF_HUB_CACHE/" "s3://$DSQA_BUCKET/hf-cache/" --only-show-errors
aws s3 ls "s3://$DSQA_BUCKET/hf-cache/" --summarize --human-readable --recursive | tail -2

# 5. Disco limpio antes de crear la AMI: pesos, modelos y datos viven en S3 (ADR-15)
rm -rf "$HF_HOME" /opt/dsqa/repo/data /opt/dsqa/repo/models
